const shapeChart = node({
  type: 'n8n-nodes-base.code',
  version: 2,
  config: {
    name: 'Shape chart rows',
    parameters: {
      mode: 'runOnceForAllItems',
      language: 'javaScript',
      jsCode: `// One row per sound for trending_songs. A sound counts as a released song only
// when TikTok matched it to a real track with a title and artist; original
// sounds are kept for the record but never picked.
const settings = $('Read picker settings').first().json;
const resp = $input.first().json || {};
const list = Array.isArray(resp.items) ? resp.items : (Array.isArray(resp.data) ? resp.data : []);
const isSample = settings.chart_source !== 'tokchart';
const chartDate = $now.setZone('America/New_York').toFormat('yyyy-MM-dd');
const names = (a) => (Array.isArray(a) ? a : []).map((x) => (typeof x === 'string' ? x : (x && x.name) || '')).filter(Boolean);
const rows = [];
for (const it of list) {
  if (!it || it.sound_id === undefined || it.sound_id === null) continue;
  const m = it.matched_song || null;
  const artists = m ? names(m.artists) : [];
  const title = m && m.title ? String(m.title) : (it.title || null);
  const released = !!m && it.content_type !== 'original' && artists.length > 0 && !!(m && m.title);
  rows.push({
    chart_date: chartDate,
    country: settings.country,
    sound_id: String(it.sound_id),
    chart_index: it.chart_index ?? null,
    song: released ? artists.join(', ') + ' - ' + m.title : null,
    title,
    artists,
    content_type: it.content_type || null,
    is_released_song: released,
    tokchart_score: it.tokchart_score ?? null,
    ugc_videos_total: it.ugc_videos_total ?? null,
    ugc_videos_per_day: it.ugc_videos_per_day ?? null,
    ugc_videos_acceleration: it.ugc_videos_acceleration ?? null,
    genres: m && Array.isArray(m.genres) ? m.genres : [],
    tiktok_url: it.tiktok_url || null,
    spotify_url: it.spotify_url || null,
    is_sample: isSample,
    raw: it
  });
}
if (!rows.length) throw new Error('The chart came back empty (' + (isSample ? 'sample' : 'TokChart') + '). Posts keep their songs.');
return [{ json: { rows, count: rows.length, released: rows.filter((r) => r.is_released_song).length } }];`
    },
    output: [{ rows: [{ chart_date: '2026-11-17', country: 'US', sound_id: 'sample-1', song: 'Miley Cyrus - Flowers', is_released_song: true }], count: 31, released: 28 }]
  }
});

const readChoices = node({
  type: 'n8n-nodes-base.code',
  version: 2,
  config: {
    name: 'Read AI choices',
    parameters: {
      mode: 'runOnceForEachItem',
      language: 'javaScript',
      jsCode: `// A failed or garbled AI answer is not an error: the post keeps its song
// and the reason is logged.
const req = $('Build pick request').item.json;
const r = $json || {};
let choices = [];
let noFit = null;
try {
  let c = r.choices[0].message.content;
  c = String(c).replace(/^\`\`\`(?:json)?\\s*/i, '').replace(/\`\`\`\\s*$/, '').trim();
  const a = c.indexOf('{'), b = c.lastIndexOf('}');
  const o = JSON.parse(a >= 0 && b > a ? c.slice(a, b + 1) : c);
  choices = (Array.isArray(o.choices) ? o.choices : [])
    .filter((x) => x && x.song)
    .slice(0, 3)
    .map((x) => ({ song: String(x.song), reason: x.reason ? String(x.reason).slice(0, 300) : null }));
  noFit = o.no_fit_reason ? String(o.no_fit_reason).slice(0, 300) : null;
} catch (e) {
  const err = r.error ? (r.error.message || JSON.stringify(r.error)) : e.message;
  noFit = 'the AI gave no usable answer: ' + String(err).slice(0, 200);
}
return { json: {
  p_content_type: req.post.content_type,
  p_content_id: req.post.content_id,
  p_choices: choices,
  p_model: r.model || null,
  p_no_fit_reason: noFit
} };`
    },
    output: [{ p_content_type: 'jealousy_carousel', p_content_id: 'JC-1', p_choices: [{ song: 'Miley Cyrus - Flowers', reason: 'moving on' }], p_model: 'qwen/qwen3.8-flash', p_no_fit_reason: null }]

const note = sticky(
  '## Trending Song Picker\n' +
  'Gives each upcoming Character 6 post a song that is trending on TikTok AND fits its story, before the 01:15 ET hand-out to the To-do list.\n\n' +
  'All rules live in Supabase (migration 20261117120000_music_picker.sql). Switches in `music_picker_settings`:\n' +
  '- `dry_run` true: picks are only logged in `music_picks`; no post changes.\n' +
  '- `chart_source` sample until the TokChart subscription, then tokchart (and attach the TokChart API credential).\n\n' +
  'Plan: docs/MUSIC-PICKER-PLAN.md',
  [chartTrigger, pickTrigger],
  { color: 4 }
);

export default workflow('music-trending-song-picker', '[Music] Trending Song Picker')
  .add(chartTrigger)
  .to(readSettings)
  .to(isLiveChart
    .onTrue(fetchTokchart.to(shapeChart))
    .onFalse(sampleChart.to(shapeChart)))
  .add(shapeChart)
  .to(saveChart)
  .to(songsToDescribe)
  .to(buildDescribe)
  .to(describeAI)
  .to(shapeNotes)
  .to(saveNotes)
  .add(pickTrigger)
  .to(findPosts)
  .to(songPool)
  .to(pairPosts)
  .to(eachPost
    .onEachBatch(buildPick.to(pickAI).to(readChoices).to(applyPick).to(nextBatch(eachPost))))
  .add(note)
  .group('Chart pull', [readSettings, isLiveChart, fetchTokchart, sampleChart, shapeChart, saveChart], {
    description: "Fetches today's US trending sounds (sample list until the TokChart subscription) and saves them in trending_songs."
  })
  .group('Song notes', [songsToDescribe, buildDescribe, describeAI, shapeNotes, saveNotes], {
    description: 'Describes each new song once (mood, themes, what it suits) so stories can be matched to it.'
  })
  .group('Pick songs', [findPosts, songPool, pairPosts, eachPost, buildPick, pickAI, readChoices, applyPick], {
    description: 'For each upcoming post, the AI ranks fitting songs; Supabase applies the first that passes every rule.'
  });
